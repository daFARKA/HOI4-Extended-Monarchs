# Germany proposes unification
country_event = { 
	id = em.1
	title = em.1.t
	desc = em.1.desc
	picture = GFX_report_event_generic_sign_treaty2

	is_triggered_only = yes

	option = { 
		name = em.1.a
		ai_chance = {
			base = 20
			modifier = {
				factor = 2
				has_government = FROM
			}
			modifier = {
				factor = 2
				strength_ratio = {
				    tag = FROM
				    ratio < 0.75
				}
			}
			modifier = {
				factor = 2
				strength_ratio = {
				    tag = FROM
				    ratio < 0.5
				}
			}
		}

		FROM = { # FROM Accepts
			country_event = { 
				id = em.2
				hours = 5 
				random_hours = 6
			}
		}

		effect_tooltip = {
			FROM = {
				annex_country = {
					target = ROOT
					transfer_troops = yes
				}
			}
			ROOT = {
				every_core_state = {		
					add_core_of = FROM
				}
			}
		}

		custom_effect_tooltip = generic_skip_one_line_tt
		custom_effect_tooltip = GAME_OVER_TT

	}

	# We will not compromise on our independence!
	option = {
		name = em.1.b
		ai_chance = {
			base = 2
			modifier = {
				factor = 5
				strength_ratio = {
				    tag = FROM
				    ratio > 1
				}
			}
			modifier = {
				factor = 5
				strength_ratio = {
				    tag = FROM
				    ratio > 1.25
				}
			}
			modifier = {
				factor = 0
				is_subject_of = FROM
			}

			modifier = {
				factor = 200 #needs to be this big to cancel out effects in option a
				FROM = {
					is_faction_leader = yes
				}
				any_other_country = {
					NOT = { original_tag = FROM }
					is_in_faction_with = FROM
				}
			}
		}

		add_war_support = 0.1

		custom_effect_tooltip = generic_skip_one_line_tt

		FROM = { # FROM Rejects
			country_event = {
				id = em.3
				hours = 5 
				random_hours = 6
			}
		}

	}
}
